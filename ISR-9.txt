import cv2
import os
import numpy as np
import matplotlib.pyplot as plt
import imagehash
from PIL import Image

# Give your images folder path here
folder = r"C:\Users\ishaa\OneDrive\Desktop\TextDocuments\images"

# Read image files
files = [f for f in os.listdir(folder)
         if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

print("Total Images:", len(files))

if len(files) == 0:
    print("No images found. Check folder path.")
else:

    # Feature extraction
    def extract_features(path):
        img = cv2.imread(path)

        hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

        # Colour histogram
        color = cv2.calcHist(
            [hsv], [0, 1, 2], None,
            [8, 8, 8], [0, 180, 0, 256, 0, 256]
        )
        color = cv2.normalize(color, color).flatten()

        # Texture extraction
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        edges = cv2.Sobel(gray, cv2.CV_64F, 1, 1)

        texture, _ = np.histogram(
            edges, bins=32, range=(-255, 255)
        )
        texture = texture / (np.linalg.norm(texture) + 1e-10)

        return color, texture

    # Select input image
    query_image = os.path.join(folder, files[0])
    query_color, query_texture = extract_features(query_image)

    print("Input Image:", files[0])

    # Plot histograms
    plt.figure(figsize=(10, 4))

    plt.subplot(1, 2, 1)
    plt.plot(query_color)
    plt.title("Colour Histogram")

    plt.subplot(1, 2, 2)
    plt.plot(query_texture)
    plt.title("Texture Histogram")

    plt.show()

    # Calculate similarity
    results = []

    query_hash = imagehash.average_hash(
        Image.open(query_image)
    )

    for file in files:
        path = os.path.join(folder, file)

        color, texture = extract_features(path)

        color_distance = cv2.compareHist(
            query_color.astype('float32'),
            color.astype('float32'),
            cv2.HISTCMP_BHATTACHARYYA
        )

        texture_distance = np.linalg.norm(
            query_texture - texture
        )

        img_hash = imagehash.average_hash(Image.open(path))
        hash_distance = query_hash - img_hash

        score = color_distance + texture_distance + hash_distance / 64

        results.append((file, score))

    # Sort results
    results.sort(key=lambda x: x[1])

    # Display similar images
    print("\nSimilar Images:")

    plt.figure(figsize=(12, 4))

    for i, (file, score) in enumerate(results[:5]):
        print(file, "Score:", round(score, 3))

        img = Image.open(os.path.join(folder, file))

        plt.subplot(1, 5, i + 1)
        plt.imshow(img)
        plt.title(file)
        plt.axis("off")

    plt.tight_layout()
    plt.show()
