import tempfile
from collections import defaultdict
from whoosh.index import create_in
from whoosh.fields import Schema, TEXT, ID
from whoosh.qparser import QueryParser
from whoosh import scoring
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

DOCS = {
    "d1": "Python is a popular programming language for data science",
    "d2": "Java is a programming language used for enterprise applications",
    "d3": "Information retrieval systems index documents using inverted files",
    "d4": "Python libraries like Whoosh help build search engines quickly",
    "d5": "Elasticsearch and Lucene are powerful search engines built on inverted indexes",
}

inv = defaultdict(set)
for did, text in DOCS.items():
    for w in text.lower().split():
        inv[w].add(did)

print("=== Inverted file ===")
for t in ["python", "search", "inverted", "language"]:
    print(t, "->", sorted(inv[t]))
print("python AND search :", sorted(inv["python"] & inv["search"]))
print("python OR java    :", sorted(inv["python"] | inv["java"]))
print("language NOT java :", sorted(inv["language"] - inv["java"]))

schema = Schema(doc_id=ID(stored=True, unique=True), content=TEXT(stored=True))
ix = create_in(tempfile.mkdtemp(), schema)
w = ix.writer()
for did, text in DOCS.items():
    w.add_document(doc_id=did, content=text)
w.commit()

print("\n=== Whoosh Boolean search ===")
with ix.searcher() as s:
    parser = QueryParser("content", ix.schema)
    for q in ["python AND search", "language NOT java", "python OR java", "inverted"]:
        print(f"'{q}':", [r["doc_id"] for r in s.search(parser.parse(q))])

print("\n=== Whoosh TF-IDF ranking ===")
with ix.searcher(weighting=scoring.TF_IDF()) as s:
    q = QueryParser("content", ix.schema).parse("python search engine")
    for r in s.search(q):
        print(r["doc_id"], round(r.score, 3), r["content"])

print("\n=== Cosine similarity (vector space model) ===")
ids = list(DOCS)
tv = TfidfVectorizer()
M = tv.fit_transform(DOCS.values())
sims = cosine_similarity(tv.transform(["python search engine"]), M)[0]
for i in sims.argsort()[::-1]:
    print(ids[i], round(sims[i], 3))


def elasticsearch_demo():
    from elasticsearch import Elasticsearch
    es = Elasticsearch("http://localhost:9200")
    for did, text in DOCS.items():
        es.index(index="ir_demo", id=did, document={"content": text})
    es.indices.refresh(index="ir_demo")
    r = es.search(index="ir_demo", query={"bool": {
        "must": [{"match": {"content": "python"}}],
        "must_not": [{"match": {"content": "java"}}]}})
    print([h["_id"] for h in r["hits"]["hits"]])
