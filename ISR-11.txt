# Simple Search Engine

# Step 1: Website Pages
pages = {
    1: ["Python Programming", "Python is used for web development and data science"],
    2: ["Java Programming", "Java is used to develop applications"],
    3: ["Web Development", "HTML CSS and JavaScript create websites"],
    4: ["Database", "Database stores and manages data"],
    5: ["Machine Learning", "Machine learning uses data and algorithms"],
    6: ["Python Data Science", "Python is used in machine learning"]
}

# Step 2: Create Inverted Index
index = {}

for id, page in pages.items():
    words = (page[0] + " " + page[1]).lower().split()

    for word in words:
        if word not in index:
            index[word] = []
        index[word].append(id)

print("Index Created Successfully!")

# Step 3: Search Keyword
query = input("Enter keyword to search: ").lower()

results = {}

for word in query.split():
    if word in index:
        for id in index[word]:
            results[id] = results.get(id, 0) + 1

# Step 4: Display Results
if results:
    print("\nSearch Results:")

    for id in sorted(results, key=results.get, reverse=True):
        print("\nPage:", pages[id][0])
        print("Content:", pages[id][1])
        print("Score:", results[id])
else:
    print("No results found!")
